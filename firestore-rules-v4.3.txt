rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function signedIn() { return request.auth != null; }
    function profile() { return get(/databases/$(database)/documents/userProfiles/$(request.auth.uid)).data; }
    function hasRole(roles) { return signedIn() && profile().role in roles; }

    match /userProfiles/{uid} {
      allow read: if signedIn() && (request.auth.uid == uid || hasRole(['admin']));
      allow create, update, delete: if hasRole(['admin']);
    }

    match /students/{id} {
      allow read: if hasRole(['admin','staff','scanner']);
      allow create, update, delete: if hasRole(['admin']);
    }

    match /pickupQueue/{id} {
      allow read: if hasRole(['admin','staff','scanner','display']);
      allow create: if hasRole(['admin','staff','scanner']);
      allow update, delete: if hasRole(['admin','staff']);
    }

    match /pickupHistory/{id} {
      allow read: if hasRole(['admin','staff','display']);
      allow create, update, delete: if hasRole(['admin','staff']);
    }

    match /auditLog/{id} {
      allow read: if hasRole(['admin']);
      allow create: if hasRole(['admin','staff','scanner']);
      allow update, delete: if false;
    }
  }
}
